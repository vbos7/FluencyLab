Parte 2 — Salvar edição (Client Component)

Em edit-profile-dialog.tsx, o handleSave hoje é um // TODO: chamada de API. Substitua:

import { apiClient } from "@/app/_lib/api"

async function handleSave() {
    try {
        await apiClient.put("/profile.php", { name: form.name, email: form.email })
        setOpen(false)
        window.location.reload()  // recarrega para mostrar dados atualizados
    } catch (err) {
        console.error("Erro ao salvar perfil:", err)
    }
}
