Parte 2 — Salvar edição (Client Component)

Em edit-profile-dialog.tsx, adicione a prop initialPhone, envie todos os campos (incluindo os de senha) e mostre os erros que o backend devolver.

frontend/app/_components/profile/edit-profile-dialog.tsx
import axios from "axios"
import { apiClient } from "@/app/_lib/api"

// ...dentro do componente:
const [errors, setErrors] = useState<string[]>([])
const [saving, setSaving] = useState(false)

async function handleSave() {
    setErrors([])
    setSaving(true)
    try {
        await apiClient.put("/profile.php", {
            name: form.nome,
            email: form.email,
            phone: form.telefone,
            // O backend só troca a senha se new_password vier preenchido
            current_password: form.senhaAtual,
            new_password: form.novaSenha,
            new_password_confirmation: form.confirmarSenha,
        })
        setOpen(false)
        window.location.reload()  // recarrega para mostrar dados atualizados
    } catch (err) {
        // O PHP devolve { errors: [...] } com status 422
        if (axios.isAxiosError(err) && err.response?.data?.errors) {
            setErrors(err.response.data.errors)
        } else {
            setErrors(["Não foi possível salvar. Tente novamente."])
        }
    } finally {
        setSaving(false)
    }
}

Por que enviar os campos de senha sempre? O backend checa se new_password está vazio — se estiver, ele ignora a troca de senha e só atualiza nome/email/telefone. Assim o mesmo endpoint serve pros dois casos, sem precisar de duas rotas.
