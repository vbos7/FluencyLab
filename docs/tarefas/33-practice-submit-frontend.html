Submeter e exibir a correção por IA

A função handleVerify em practice-controller.tsx hoje calcula a correção localmente (Levenshtein — o protótipo de apresentação). Aqui você troca pela chamada real ao PHP que chama a OpenAI.

Função adaptadora (converte resposta da IA para o formato da tela)

type AiFeedback = {
    is_correct: boolean; score: number; overall_comment: string
    mistakes: { type: string; original: string; suggestion: string; explanation_pt: string }[]
    corrected_sentence: string; positive_points: string[]
}

function adaptAiFeedback(ai: AiFeedback, xp: number): Feedback {
    const status = ai.score >= 95 ? "perfect" : ai.score >= 70 ? "good" : ai.score >= 40 ? "partial" : "needs_work"
    const titles = { perfect: "Perfeito!", good: "Muito bem!", partial: "Quase lá!", needs_work: "Continue praticando" }
    return {
        status,
        title: titles[status],
        message: ai.overall_comment,
        corrections: ai.mistakes.map((m) => ({ wrong: m.original, correct: m.suggestion, explanation: m.explanation_pt })),
        xp,
    }
}

handleVerify assíncrona

import { apiClient } from "@/app/_lib/api"

const [loading, setLoading] = useState(false)

const handleVerify = async () => {
    if (!answer.trim()) return
    setLoading(true)

    try {
        const response = await apiClient.post("/practice/check-answer.php", {
            phrase_id: phrase.id,
            answer,
        })
        const fb = adaptAiFeedback(response.data.feedback, response.data.xp_earned)
        setFeedback(fb)
        setEarnedXp(fb.xp)
        setShowXpToast(true)
        setTimeout(() => setShowXpToast(false), 2000)
    } catch (err) {
        console.error("Erro ao corrigir:", err)
    } finally {
        setLoading(false)
    }
}

Encontre o botão que chama handleVerify (provavelmente em answer-form.tsx) e adicione disabled={loading} + texto "Corrigindo..." enquanto aguarda.
