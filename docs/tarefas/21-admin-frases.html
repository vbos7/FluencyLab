Listar frases (admin)

Listagem administrativa das frases com contagem de tentativas — inclui um LEFT JOIN com COUNT, padrão que você já viu na tarefa 16.

O arquivo

backend/api/admin/phrases.php
<?php
require_once __DIR__ . '/../cors.php';
require_once __DIR__ . '/../db.php';

if ($_SERVER['REQUEST_METHOD'] !== 'GET') { json_out(['error' => 'Método não permitido'], 405); exit; }

if (!isset($_SESSION['user_id']) || $_SESSION['role'] !== 'admin') {
    json_out(['error' => 'Acesso restrito a administradores'], 403);
    exit;
}

$stmt = $pdo->query("
    SELECT p.*, COUNT(a.id) AS total_attempts
    FROM phrases p
    LEFT JOIN attempts a ON a.phrase_id = p.id
    GROUP BY p.id
    ORDER BY total_attempts DESC
");
json_out($stmt->fetchAll());
