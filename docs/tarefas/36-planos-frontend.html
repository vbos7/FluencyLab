Checkout (substituir o setTimeout fake)

Em checkout-modal.tsx, adicione uma prop planId e troque o setTimeout:

import { apiClient } from "@/app/_lib/api"

async function handleSubmit() {
    setLoading(true)
    try {
        await apiClient.post("/plans-subscribe.php", { plan_id: planId })
        setSuccess(true)
    } catch {
        console.error("Erro ao assinar plano")
    } finally {
        setLoading(false)
    }
}
